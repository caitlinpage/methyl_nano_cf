rrbs_plot5 <- function(read_data, read_ids, cpgs = cg_sites, jit_amount = 0.1) {
  set.seed(42) # For reproducibility
  # Calculate a single jitter amount per row (segment)
  read_data <- read_data %>% dplyr::filter(read_id %in% read_ids)
  jitter_amount <- runif(nrow(read_data), -jit_amount, jit_amount)
  read_data$alpha_jit <- read_data$alpha + jitter_amount
  read_data <- data.frame(read_data)

  # Add jittered coordinates to new columns in the data frame
  long_data <- read_data %>%
    mutate(meth_status = strsplit(meth_pattern, ""), start = strsplit(genom_positions, ",")) %>%
    unnest(meth_status, start) %>%
    mutate(start = as.double(start))
  
  plot <- read_data %>%
    ggplot() +
    geom_segment(aes(x = read_start, xend = read_end, y = alpha_jit, yend = alpha_jit), alpha = 0.2) +
    geom_point(data = long_data,
               aes(x = start, y = alpha_jit, fill = meth_status), shape = 21, color = "black") +
    scale_fill_manual(values = c("U" = "white", "M" = "black")) +
    facet_wrap(~ sample)
  plot
}
