process_wgbs_short <- function(bismark, bed = NULL) {
  read <- bismark
  names(read) <- c("read_id", "strand", "seqnames", "start", "meth_status")
  read$end <- read$start

  read_overlap <- read
  read_overlap[, meth_status := ifelse(meth_status == "Z", "M", "U")]
  read_overlap[, meth_pattern := paste0(meth_status, collapse = ""), by = read_id]
  read_overlap[, num_meth := str_count(meth_pattern, pattern = "M")]
  read_overlap[, total := str_length(meth_pattern)]
  read_overlap[, alpha := num_meth/total]
  read_overlap[, genom_positions := paste0(start, collapse = ","), by = read_id]

  read_overlap <- read_overlap %>% .[order(.$read_id),]

  read_overlap[, read_start := min(start), by = read_id]
  read_overlap[, read_end := max(start), by = read_id]

  read_overlap
}

# --- Analysis function ---
run_analysis <- function(combo) {
  cols <- c(combo$g1, combo$g2)
  group <- c(1, 1, 1, 2, 2, 2)

  y <- DGEList(
    as.matrix(dtu_optile[, cols]),
    group = group,
    genes = dtu_optile[, c("gene_id", "transcript")]
  )
  design <- model.matrix(~ group)
  fit <- glmQLFit(y, design, robust = TRUE)
  ds  <- diffSpliceDGE(fit, coef = 2, geneid = "gene_id", exonid = "transcript")

  topSpliceDGE(ds, test = "gene", number = Inf) %>%
    mutate(
      region_id = gene_id,
      excluded  = combo$excluded,
      group1    = paste(combo$g1, collapse = ","),
      group2    = paste(combo$g2, collapse = ",")
    ) %>%
    separate_wider_delim(gene_id, delim = ":", names = c("seqnames", "gene_id")) %>%
    mutate(seqnames = gsub("chrchr", "chr", seqnames)) %>%
    separate_wider_delim(gene_id, delim = "-", names = c("start", "end")) %>%
    mutate(
      start    = as.double(start),
      end      = as.double(end),
      dtu_rank = row_number()
    )
}

rrbs_plot5 <- function(read_data, read_ids, cpgs = cg_sites, jit_amount = 0.1) {
  set.seed(42) # For reproducibility
  # Calculate a single jitter amount per row (segment)
  read_data <- read_data %>% dplyr::filter(read_id %in% read_ids)
  jitter_amount <- runif(nrow(read_data), -jit_amount, jit_amount)
  read_data$alpha_jit <- read_data$alpha + jitter_amount
  read_data <- data.frame(read_data)

  # Add jittered coordinates to new columns in the data frame
  long_data <- read_data %>%
    mutate(meth_status = strsplit(meth_pattern, ""), start = strsplit(genom_positions, ",")) %>%
    unnest(meth_status, start) %>%
    mutate(start = as.double(start))
  
  plot <- read_data %>%
    ggplot() +
    geom_segment(aes(x = read_start, xend = read_end, y = alpha_jit, yend = alpha_jit), alpha = 0.2) +
    geom_point(data = long_data,
               aes(x = start, y = alpha_jit, fill = meth_status), shape = 21, color = "black") +
    scale_fill_manual(values = c("U" = "white", "M" = "black")) +
    facet_wrap(~ sample)
  
  distrib <- read_data %>%
    ggplot(aes(x = alpha, colour = alpha)) +
    geom_density()
  
  list(plot, distrib)
}

run_analysis <- function(combo) {
  cols <- c(combo$g1, combo$g2)
  group <- c(1, 1, 1, 2, 2, 2)

  y <- DGEList(
    as.matrix(dtu_optile[, cols]),
    group = group,
    genes = dtu_optile[, c("gene_id", "transcript")]
  )
  design <- model.matrix(~ group)
  fit <- glmQLFit(y, design, robust = TRUE)
  ds  <- diffSpliceDGE(fit, coef = 2, geneid = "gene_id", exonid = "transcript")

  topSpliceDGE(ds, test = "exon", number = Inf) %>%
    mutate(
      region_id = gene_id,
      excluded  = combo$excluded,
      group1    = paste(combo$g1, collapse = ","),
      group2    = paste(combo$g2, collapse = ",")
    ) %>%
    separate_wider_delim(gene_id, delim = ":", names = c("seqnames", "gene_id")) %>%
    mutate(seqnames = gsub("chrchr", "chr", seqnames)) %>%
    separate_wider_delim(gene_id, delim = "-", names = c("start", "end")) %>%
    mutate(
      start    = as.double(start),
      end      = as.double(end),
      dtu_rank = row_number()
    )
}
